     1	"""Dependency-free localhost sample. Restart to reset its disposable data."""
     2	
     3	import argparse
     4	import json
     5	import os
     6	import threading
     7	from datetime import datetime, timezone
     8	from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
     9	from pathlib import Path
    10	from urllib.parse import urlsplit
    11	
    12	from domain import ApiError, Store
    13	
    14	STATIC = Path(__file__).parent / "static"
    15	
    16	
    17	def create_server(port=0, audit_path=None):
    18	    store = Store()
    19	    lock = threading.Lock()
    20	
    21	    class Handler(BaseHTTPRequestHandler):
    22	        def log_message(self, *args):
    23	            pass
    24	
    25	        def respond(self, status, payload, actor=None, request_body=None, content_type="application/json"):
    26	            data = json.dumps(payload).encode() if content_type == "application/json" else payload
    27	            self.send_response(status)
    28	            self.send_header("Content-Type", content_type)
    29	            self.send_header("Content-Length", str(len(data)))
    30	            self.end_headers()
    31	            self.wfile.write(data)
    32	            if audit_path:
    33	                record = {"time": datetime.now(timezone.utc).isoformat(), "method": self.command,
    34	                          "path": urlsplit(self.path).path, "actor": actor,
    35	                          "request": request_body, "status": status,
    36	                          "response": payload if content_type == "application/json" else None}
    37	                with lock, open(audit_path, "a") as out:
    38	                    out.write(json.dumps(record) + "\n")
    39	
    40	        def user(self):
    41	            token = self.headers.get("Authorization", "")
    42	            identities = {"Bearer demo-alice": "alice", "Bearer demo-bob": "bob"}
    43	            if token not in identities:
    44	                raise ApiError(401, "Authentication required")
    45	            return identities[token]
    46	
    47	        def do_GET(self):
    48	            path = urlsplit(self.path).path
    49	            if path in ("/", "/cart"):
    50	                return self.respond(200, (STATIC / "index.html").read_bytes(), content_type="text/html")
    51	            if path in ("/app.js", "/style.css"):
    52	                kind = "text/javascript" if path.endswith(".js") else "text/css"
    53	                return self.respond(200, (STATIC / path[1:]).read_bytes(), content_type=kind)
    54	            if path == "/favicon.ico":
    55	                return self.respond(204, b"", content_type="image/x-icon")
    56	            if path == "/health":
    57	                return self.respond(200, {"status": "UP", "version": "sample-1"})
    58	            actor = None
    59	            try:
    60	                actor = self.user()
    61	                if path == "/api/v1/products":
    62	                    return self.respond(200, store.catalog(), actor)
    63	                if path == "/api/v1/cart":
    64	                    return self.respond(200, store.cart(actor), actor)
    65	                raise ApiError(404, "Not found")
    66	            except ApiError as error:
    67	                self.respond(error.status, {"error": error.message}, actor)
    68	
    69	        def do_PUT(self):
    70	            actor, body = None, None
    71	            try:
    72	                actor = self.user()
    73	                path = urlsplit(self.path).path
    74	                if not path.startswith("/api/v1/cart/items/"):
    75	                    raise ApiError(404, "Not found")
    76	                length = int(self.headers.get("Content-Length", 0))
    77	                if length > 4096:
    78	                    raise ApiError(413, "Request too large")
    79	                body = json.loads(self.rfile.read(length))
    80	                if not isinstance(body, dict):
    81	                    raise ApiError(400, "Expected a JSON object")
    82	                result = store.update(actor, int(path.rsplit("/", 1)[1]), body.get("quantity"))
    83	                self.respond(200, result, actor, body)
    84	            except (ValueError, TypeError):
    85	                self.respond(400, {"error": "Invalid request"}, actor, body)
    86	            except ApiError as error:
    87	                self.respond(error.status, {"error": error.message}, actor, body)
    88	
    89	    return ThreadingHTTPServer(("127.0.0.1", port), Handler)
    90	
    91	
    92	if __name__ == "__main__":
    93	    parser = argparse.ArgumentParser()
    94	    parser.add_argument("--port", type=int, default=0)
    95	    parser.add_argument("--ready", type=Path)
    96	    parser.add_argument("--audit", type=Path)
    97	    args = parser.parse_args()
    98	    server = create_server(args.port, args.audit)
    99	    ready = {"pid": os.getpid(), "base_url": f"http://127.0.0.1:{server.server_port}"}
   100	    if args.ready:
   101	        args.ready.write_text(json.dumps(ready))
   102	    print(json.dumps(ready), flush=True)
   103	    server.serve_forever()
