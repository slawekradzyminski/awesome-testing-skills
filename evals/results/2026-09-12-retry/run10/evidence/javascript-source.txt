     1	const identity = document.querySelector('#identity');
     2	const form = document.querySelector('#quantity-form');
     3	const quantity = document.querySelector('#quantity');
     4	const summary = document.querySelector('#summary');
     5	const feedback = document.querySelector('#feedback');
     6	let current;
     7	
     8	const headers = () => ({Authorization: `Bearer demo-${identity.value}`, 'Content-Type': 'application/json'});
     9	
    10	async function refresh() {
    11	  const response = await fetch('/api/v1/cart', {headers: headers()});

    14	}
    15	
    16	document.querySelector('#edit').addEventListener('click', () => {
    17	  quantity.value = current.items[0]?.quantity ?? 0;
    18	  form.hidden = false;

    20	  quantity.focus();
    21	});
    22	
    23	document.querySelector('#cancel').addEventListener('click', () => {
    24	  form.hidden = true;

    27	});
    28	
    29	form.addEventListener('submit', async event => {
    30	  event.preventDefault();

    32	    method: 'PUT', headers: headers(), body: JSON.stringify({quantity: Number(quantity.value)}),
    33	  });
    34	  if (response.ok) {
    35	    await refresh();
    36	    form.hidden = true;
    37	    feedback.textContent = 'Cart updated';
    38	  } else {
    39	    const body = await response.json();
    40	    feedback.textContent = body.error;
    41	  }
    42	});
    43	
    44	identity.addEventListener('change', async () => {
    45	  form.hidden = true;
    46	  feedback.textContent = '';

    48	});
